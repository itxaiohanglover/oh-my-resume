% Advanced style: trapezoid block + horizontal line
\newcommand{\omrSectionRefined}[1]{%
  \vspace{\omrSectionBefore}
  \setlength{\omrSavedLineWidth}{\linewidth}%
  \begin{tcolorbox}[
    skin=empty,
    left=12pt,
    right=5pt,
    top=1pt,
    bottom=1pt,
    width=0.20\linewidth,
    halign=left,
    valign=center,
    fontupper=\color{white}\fontsize{\omrSectionFontSize}{\omrSectionLineHeight}\selectfont\bfseries,
    frame code={
        \fill[omrSectionBg] (frame.south west) -- (frame.north west) -- ([xshift=-16pt]frame.north east) -- (frame.south east) -- cycle;
    },
    overlay={
        \fill[omrSectionBg]
          ([xshift=4pt, yshift=4pt]frame.south east) --
          ([xshift=8pt, yshift=0pt]frame.south east) --
          ({\the\dimexpr\omrSavedLineWidth-0pt\relax}, 0pt) --
          ({\the\dimexpr\omrSavedLineWidth-0pt\relax}, 4pt) --
          cycle;
    },
    before skip=0pt,
    after skip=0pt
  ]
  #1
  \end{tcolorbox}
  \vspace{\omrSectionAfter}
}

% Simple style: colored text + full-width underline
\newcommand{\omrSectionSimple}[1]{%
  \vspace{\omrSectionBefore}
  \noindent
  {\fontsize{\omrSectionFontSize}{\omrSectionLineHeight}\selectfont\bfseries\color{omrSectionBg}#1}
  \vspace{-4pt}\par\vspace{-4pt}
  \noindent\textcolor{omrSectionBg}{\rule{\linewidth}{1.2pt}}
  \par\vspace{-2pt}
}

% Classic style: original left-rule + light background
\newcommand{\omrSectionClassic}[1]{%
  \vspace{\omrSectionBefore}
  \begin{tcolorbox}[
    enhanced,
    colback=omrClassicBg,
    colframe=omrClassicBg,
    boxrule=0pt,
    frame hidden,
    overlay={\draw[omrSectionBg,line width=3pt] (frame.north west) -- (frame.south west);},
    rightrule=0pt,
    toprule=0pt,
    bottomrule=0pt,
    arc=0pt,
    boxsep=2.5pt,
    left=12pt,
    right=0pt,
    top=2.5pt,
    bottom=2.5pt,
    width=\linewidth,
    before skip=0pt,
    after skip=0pt
  ]
  {\fontsize{\omrSectionFontSize}{\omrSectionLineHeight}\selectfont\textbf{#1}}
  \end{tcolorbox}
  \vspace{\omrSectionAfter}
}

% Minimal style: text + right-side horizontal line on same line
\newcommand{\omrSectionMinimal}[1]{%
  \vspace{\omrSectionBefore}
  \noindent
  \sbox0{\fontsize{\omrSectionFontSize}{\omrSectionLineHeight}\selectfont\bfseries\color{omrSectionBg}#1}%
  \usebox0%
  \hspace{0.5em}%
  {\color{omrSectionBg}\raisebox{0.4ex}{\rule{\dimexpr\linewidth-\wd0-0.5em\relax}{1.2pt}}}%
  \par\vspace{\omrSectionAfter}
}

% Professional style: tag shape (rounded rect + triangle) + horizontal line
\newcommand{\omrSectionProfessional}[1]{%
  \vspace{\omrSectionBefore}
  \setlength{\omrSavedLineWidth}{\linewidth}%
  \begin{tcolorbox}[
    skin=empty,
    left=8pt,
    right=5pt,
    top=1pt,
    bottom=1pt,
    width=0.16\linewidth,
    halign=left,
    valign=center,
    fontupper=\color{white}\fontsize{\omrSectionFontSize}{\omrSectionLineHeight}\selectfont\bfseries,
    frame code={
        \fill[omrSectionBg, rounded corners=3pt]
          ([xshift=10pt]frame.east) [sharp corners] --
          (frame.south east) --
          (frame.south west) --
          (frame.north west) --
          (frame.north east) --
          cycle;
    },
    overlay={
        \draw[omrSectionBg, line width=1.2pt]
          ([xshift=16pt]frame.east) --
          ++({\dimexpr\omrSavedLineWidth-\omrSavedLineWidth*16/100-18pt\relax}, 0pt);
    },
    before skip=0pt,
    after skip=0pt
  ]
  #1
  \end{tcolorbox}
  \vspace{\omrSectionAfter}
}

% Router — \sectionStyle default defined in oh-my-resume.cls, overridden by renderOverrides
\newcommand{\sectioncard}[1]{%
  \ifthenelse{\equal{\sectionStyle}{simple}}{%
    \omrSectionSimple{#1}%
  }{%
    \ifthenelse{\equal{\sectionStyle}{classic}}{%
      \omrSectionClassic{#1}%
    }{%
      \ifthenelse{\equal{\sectionStyle}{premium}}{%
        \omrSectionPremium{#1}%
      }{%
        \ifthenelse{\equal{\sectionStyle}{minimal}}{%
          \omrSectionMinimal{#1}%
        }{%
          \ifthenelse{\equal{\sectionStyle}{professional}}{%
            \omrSectionProfessional{#1}%
          }{%
            \omrSectionRefined{#1}%
          }%
        }%
      }%
    }%
  }%
}
