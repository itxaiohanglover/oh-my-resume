\newcommand{\omrSection}[1]{%
  \vspace{\omrSectionBefore}
  \begin{tcolorbox}[
    enhanced,
    colback=omrSectionBg,
    colframe=omrSectionBg,
    boxrule=0pt,
    frame hidden,
    overlay={\draw[omrSectionRule,line width=3pt] (frame.north west) -- (frame.south west);},
    rightrule=0pt,
    toprule=0pt,
    bottomrule=0pt,
    arc=0pt,
    boxsep=2.5pt,
    left=12pt,
    right=0pt,
    top=2.5pt,
    bottom=2.5pt,
    width=\linewidth,
    before skip=0pt,
    after skip=0pt
  ]
  {\fontsize{\omrSectionFontSize}{\omrSectionLineHeight}\selectfont\textbf{#1}}
  \end{tcolorbox}
  \vspace{\omrSectionAfter}
}
