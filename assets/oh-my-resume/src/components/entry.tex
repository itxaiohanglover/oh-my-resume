\newcommand{\omrDatedEntry}[2]{%
  \vspace{\omrEntryBefore}%
  \noindent
  \begingroup
  \fontsize{\omrEntryFontSize}{\omrEntryLineHeight}\selectfont
  \makebox[\dimexpr\linewidth-\omrEntryDateWidth-1em\relax][l]{\bfseries #1}%
  \hfill
  \makebox[\omrEntryDateWidth][r]{\normalfont\color{omrTextMuted}#2}%
  \par
  \endgroup
  \vspace{\omrEntryAfter}%
}

  \rlap{\bfseries #1\hspace{1em}}%
  \hfill
  \makebox[0pt][c]{\bfseries #2}%
  \hfill
  \llap{\hspace{1em}\normalfont\color{omrTextMuted}#3}%
  \par
  \endgroup
  \vspace{\omrEntryAfter}%
}
