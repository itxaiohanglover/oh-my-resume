\newcommand{\omrHeader}[4]{%
  \noindent
  \ifthenelse{\isempty{#4}}{}{%
    \rlap{\hspace{-2mm}\raisebox{4mm}{\includegraphics[height=\omrLogoHeight,keepaspectratio]{#4}}}%
  }%
  \ifthenelse{\isempty{#3}}{%
    \begin{minipage}[t]{\linewidth}
      \vspace{0pt}
      \omrName{#1}
      #2
    \end{minipage}%
  }{%
    \begin{minipage}[c][\omrPhotoHeight][c]{\linewidth}
      \centering
      \omrName{#1}
      #2
    \end{minipage}%
    \llap{%
      \begin{minipage}[c][\omrPhotoHeight][c]{\dimexpr\omrPhotoWidth+\omrPhotoRightInset\relax}
        \raggedleft
        \includegraphics[height=\omrPhotoHeight,keepaspectratio]{#3}
      \end{minipage}%
    }%
  }%
}
