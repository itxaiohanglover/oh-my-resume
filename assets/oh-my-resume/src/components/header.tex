\newcommand{\omrHeaderLogo}[1]{%

}

    \begin{minipage}[t]{\linewidth}
      \vspace{0pt}
      \omrHeaderText{#1}{#2}
    \end{minipage}%
  }{%
    \begin{minipage}[c]{\linewidth}
      \vspace{0pt}
      \omrHeaderText{#1}{#2}
    \end{minipage}%
    \smash{\llap{%
      \begin{minipage}[c]{\dimexpr\omrPhotoWidth+\omrPhotoRightInset\relax}

        \includegraphics[height=\omrPhotoHeight,keepaspectratio]{#3}
      \end{minipage}%
    }}%
  }%
}
